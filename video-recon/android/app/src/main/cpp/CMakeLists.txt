cmake_minimum_required(VERSION 3.22)
project(livescan CXX)
set(CMAKE_CXX_STANDARD 17)
set(OV_ROOT "/home/charalambos/ai/phone/ov_android" CACHE PATH "OpenVINS sources patched for Android")
set(OV_LIB "${OV_ROOT}/build/libopenvins.a" CACHE FILEPATH "")
set(OPENCV_SDK "/home/charalambos/ai/phone/OpenCV-android-sdk" CACHE PATH "")
set(THIRD_PARTY_INCLUDE "/home/charalambos/ai/vio/env/include" CACHE PATH "Eigen and Boost headers")
set(OpenCV_DIR "${OPENCV_SDK}/sdk/native/jni")
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs features2d video calib3d)
add_library(livescan SHARED livescan.cpp tsdf.cpp)
target_include_directories(livescan PRIVATE ${OpenCV_INCLUDE_DIRS} ${THIRD_PARTY_INCLUDE} ${THIRD_PARTY_INCLUDE}/eigen3 ${OV_ROOT}/ov_core ${OV_ROOT}/ov_init ${OV_ROOT}/ov_msckf)
target_compile_options(livescan PRIVATE -O2 -Wno-deprecated-declarations -include cassert)
target_compile_definitions(livescan PRIVATE ROS_AVAILABLE=0)
target_link_libraries(livescan ${OV_LIB} ${OpenCV_LIBS} android log)
