cmake_minimum_required(VERSION 3.22)
project(livescan C CXX)
set(CMAKE_CXX_STANDARD 17)
set(OV_ROOT "$ENV{HOME}/ai/phone/ov_android" CACHE PATH "OpenVINS sources patched for Android")
set(OPENCV_SDK "$ENV{HOME}/ai/phone/OpenCV-android-sdk" CACHE PATH "OpenCV Android SDK")
get_filename_component(NATIVE_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.deps/native-headers" ABSOLUTE)
set(THIRD_PARTY_INCLUDE "${NATIVE_HEADERS}/include" CACHE PATH "Eigen and Boost headers")
if(ANDROID)
    set(OpenCV_DIR "${OPENCV_SDK}/sdk/native/jni")
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs features2d video calib3d)
include(InstrumentOpenVins.cmake)
include(HandheldInitialization.cmake)
# Build the dependency from source. A stale external archive can silently omit
# fixes, and a fresh checkout must not depend on someone else's build directory.
file(GLOB_RECURSE OV_SOURCES CONFIGURE_DEPENDS
    "${OV_ROOT}/ov_core/*.cpp" "${OV_ROOT}/ov_init/*.cpp" "${OV_ROOT}/ov_msckf/*.cpp")
foreach(replacement IN LISTS OBSERVED_OV_SOURCES)
    get_filename_component(original "${replacement}" NAME)
    string(REGEX REPLACE "^observed_" "" original "${original}")
    list(FILTER OV_SOURCES EXCLUDE REGEX "/${original}$")
endforeach()
add_library(openvins_live STATIC ${OV_SOURCES})
target_include_directories(openvins_live PRIVATE ${OpenCV_INCLUDE_DIRS}
    ${THIRD_PARTY_INCLUDE} ${THIRD_PARTY_INCLUDE}/eigen3
    ${OV_ROOT}/ov_core ${OV_ROOT}/ov_init ${OV_ROOT}/ov_msckf)
target_compile_definitions(openvins_live PRIVATE ROS_AVAILABLE=0)
target_compile_options(openvins_live PRIVATE -O2 -g0 -fPIC -Wno-deprecated-declarations -include cassert)
if(ANDROID)
    add_library(livescan SHARED livescan.cpp tsdf.cpp ${OBSERVED_OV_SOURCES})
    set(PLATFORM_LIBS android log)
else()
    # Host tests compile the same estimator patches and TSDF, without the JNI ABI.
    add_library(livescan SHARED tsdf.cpp ${OBSERVED_OV_SOURCES})
    target_include_directories(livescan PRIVATE ../../test/cpp/host)
    enable_testing()
endif()
# The shared host include prefix also contains desktop Ceres/glog. Android's
# matching headers must win; never mix desktop headers with the arm64 library.
set_property(TARGET livescan PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
get_target_property(ANDROID_CERES_INCLUDES Ceres::ceres INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(livescan BEFORE PRIVATE ${ANDROID_CERES_INCLUDES})
target_include_directories(livescan PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS} ${THIRD_PARTY_INCLUDE} ${THIRD_PARTY_INCLUDE}/eigen3 ${OV_ROOT}/ov_core ${OV_ROOT}/ov_init ${OV_ROOT}/ov_init/static ${OV_ROOT}/ov_msckf ${OV_ROOT}/ov_msckf/core ${OV_ROOT}/ov_msckf/update)
target_compile_options(livescan PRIVATE -O2 -g0 -Wno-deprecated-declarations -include cassert)
target_include_directories(livescan PRIVATE ${OV_ROOT}/ov_core/track ${OV_ROOT}/ov_core/feat)
target_compile_definitions(livescan PRIVATE ROS_AVAILABLE=0)
target_include_directories(livescan PRIVATE ${DYNAMIC_ROOT} ${DYNAMIC_ROOT}/dynamic ${OV_ROOT}/ov_init/init)
target_link_libraries(livescan openvins_live Ceres::ceres ${OpenCV_LIBS} ${PLATFORM_LIBS})

# Explicit opt-in native test executable; not packaged in the APK. It links the
# actual instrumented tracker/triangulator used by the app and runs via adb.
add_executable(feature_tracking_test EXCLUDE_FROM_ALL ../../test/cpp/feature_tracking_test.cpp)
get_target_property(FEATURE_TEST_INCLUDES livescan INCLUDE_DIRECTORIES)
target_include_directories(feature_tracking_test PRIVATE ${FEATURE_TEST_INCLUDES})
target_compile_definitions(feature_tracking_test PRIVATE ROS_AVAILABLE=0)
target_compile_options(feature_tracking_test PRIVATE -O2 -UNDEBUG -include cassert)
target_link_libraries(feature_tracking_test livescan ${OpenCV_LIBS} ${PLATFORM_LIBS})

add_executable(handheld_init_test EXCLUDE_FROM_ALL ../../test/cpp/handheld_init_test.cpp)
target_include_directories(handheld_init_test PRIVATE ${FEATURE_TEST_INCLUDES})
target_compile_definitions(handheld_init_test PRIVATE ROS_AVAILABLE=0)
target_compile_options(handheld_init_test PRIVATE -O2 -UNDEBUG -include cassert)
target_link_libraries(handheld_init_test livescan ${OpenCV_LIBS} ${PLATFORM_LIBS})

add_executable(map_relocalizer_test EXCLUDE_FROM_ALL ../../test/cpp/map_relocalizer_test.cpp)
target_include_directories(map_relocalizer_test PRIVATE ${FEATURE_TEST_INCLUDES})
target_compile_options(map_relocalizer_test PRIVATE -O2 -UNDEBUG)
target_link_libraries(map_relocalizer_test ${OpenCV_LIBS})

if(NOT ANDROID)
    add_executable(live_recording_replay EXCLUDE_FROM_ALL ../../test/cpp/live_recording_replay.cpp)
    target_include_directories(live_recording_replay PRIVATE ${FEATURE_TEST_INCLUDES})
    target_compile_definitions(live_recording_replay PRIVATE ROS_AVAILABLE=0)
    target_compile_options(live_recording_replay PRIVATE -O2 -g0 -include cassert)
    target_link_libraries(live_recording_replay livescan ${OpenCV_LIBS})
    add_executable(stationary_test ../../test/cpp/stationary_test.cpp)
    target_include_directories(stationary_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${THIRD_PARTY_INCLUDE}/eigen3)
    target_compile_options(stationary_test PRIVATE -UNDEBUG)
    add_executable(tsdf_quality_test ../../test/cpp/tsdf_quality_test.cpp tsdf.cpp)
    target_include_directories(tsdf_quality_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_options(tsdf_quality_test PRIVATE -UNDEBUG)
    foreach(test feature_tracking_test handheld_init_test map_relocalizer_test stationary_test tsdf_quality_test)
        add_test(NAME ${test} COMMAND ${test})
    endforeach()
endif()
